import random import string from telegram import Update, LabeledPrice from telegram.ext import Application, CommandHandler, PreCheckoutQueryHandler, SuccessfulPaymentHandler, ContextTypes # Mets ton token de bot ici TOKEN = "TON_TOKEN_TELEGRAM_ICI" # Fonction pour générer une clé de licence aléatoire def generate_license_key(): random_str = ''.join(random.choices(string.ascii_uppercase + string.digits, k=6)) return f"PRO-{random_str}" # Commande /start async def start(update: Update, context: ContextTypes.DEFAULT_TYPE): await update.message.reply_text( "👋 Bienvenue sur Traceur Pro.\n\n" "Pour utiliser la Mini App, cliquez sur le bouton ci-dessous.\n" "Si vous n'avez pas de licence, tapez la commande /buy pour acheter un accès." ) # Commande /buy (Envoie de la facture d'Étoiles) async def buy(update: Update, context: ContextTypes.DEFAULT_TYPE): chat_id = update.message.chat_id title = "Licence Traceur Pro" description = "Clé d'activation unique pour débloquer l'accès complet à la console de recherche." payload = f"payload_user_{chat_id}" currency = "XTR" # XTR = Étoiles Telegram prices = [LabeledPrice("Clé de licence", 500)] # 500 Étoiles await context.bot.send_invoice( chat_id=chat_id, title=title, description=description, payload=payload, provider_token="", # Laisser vide pour les Étoiles Telegram currency=currency, prices=prices ) # Validation obligatoire avant le paiement async def precheckout_callback(update: Update, context: ContextTypes.DEFAULT_TYPE): query = update.pre_checkout_query if query.invoice_payload.startswith("payload_user_"): await query.answer(ok=True) else: await query.answer(ok=False, error_message="Une erreur est survenue lors de la validation.") # Message après un paiement réussi : Envoi de la clé ! async def successful_payment_callback(update: Update, context: ContextTypes.DEFAULT_TYPE): key = generate_license_key() await update.message.reply_text( f"✅ **Paiement validé avec succès !** Merci pour votre achat.\n\n" f"Voici votre clé d'activation unique :\n" f"`{key}`\n\n" f"Copiez cette clé, ouvrez la Mini App, collez-la dans la zone d'activation et cliquez sur Activer pour débloquer toutes les fonctionnalités !" ) def main(): app = Application.builder().token(TOKEN).build() app.add_handler(CommandHandler("start", start)) app.add_handler(CommandHandler("buy", buy)) app.add_handler(PreCheckoutQueryHandler(precheckout_callback)) app.add_handler(MessageHandler(filters.SUCCESSFUL_PAYMENT, successful_payment_callback)) print("Le Bot est en ligne et prêt à encaisser dans le chat !") app.run_polling() if __name__ == '__main__': main()
